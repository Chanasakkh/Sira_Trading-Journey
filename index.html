📍 ขั้นตอนที่ 2: วางโค้ดลงใน Code.gs

คุณสามารถกดปุ่มสีฟ้าด้านล่างเพื่อ "คัดลอกโค้ดเต็มทั้งหมด" ไปวางแทนที่ไฟล์เดิมได้ทันที (รวมระบบเดิมและระบบ Feed ให้ครบถ้วนแล้ว): 

// ระบุ ID ของ Google Sheet
var SPREADSHEET_ID = "15q-WdmJS-Zc6j8_JOwX852ku51Wy6FrQ91wB1bc1iS8";

// 1. ฟังก์ชันสำหรับการอ่านข้อมูล (GET)
function doGet() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    
    // 1. ดึงข้อมูลตารางเทรด (ชีตแรก)
    var tradeSheet = ss.getSheets()[0];
    var tradeData = tradeSheet.getDataRange().getDisplayValues();
    
    // 2. ดึงข้อมูลตาราง VideoLinks (ถ้ามี)
    var videoSheet = ss.getSheetByName("VideoLinks");
    var videoData = [];
    if (videoSheet) {
      videoData = videoSheet.getDataRange().getValues();
    }

    // 3. ดึงข้อมูลตาราง CoverLinks สำหรับรูปหน้าปก (ถ้ามี)
    var coverSheet = ss.getSheetByName("CoverLinks");
    var coverData = [];
    if (coverSheet) {
      coverData = coverSheet.getDataRange().getValues();
    }

    // 4. ดึงข้อมูลตาราง Feed สำหรับบทวิเคราะห์รายวัน (ถ้ามี)
    var feedSheet = ss.getSheetByName("Feed");
    var feedData = [];
    if (feedSheet && feedSheet.getLastRow() > 1) {
      feedData = feedSheet.getDataRange().getValues();
    }

    // 5. ดึงข้อมูลตาราง FeedComments สำหรับความคิดเห็นใต้โพสต์ (ถ้ามี)
    var commentSheet = ss.getSheetByName("FeedComments");
    var commentData = [];
    if (commentSheet && commentSheet.getLastRow() > 1) {
      commentData = commentSheet.getDataRange().getValues();
    }
    
    // ส่งข้อมูลกลับไปแบบ JSON
    var result = {
      trades: tradeData,
      videoLinks: videoData,
      coverLinks: coverData,
      feedPosts: feedData,
      feedComments: commentData
    };
    
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ error: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// 2. ฟังก์ชันสำหรับการบันทึกข้อมูล และ ตรวจสอบสิทธิ์ (POST)
function doPost(e) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var action = e ? e.parameter.action : null;
    
    // --- ACTION: LOGIN (ตรวจสอบสิทธิ์เข้าใช้งาน) ---
    if (action === 'login') {
      var username = e.parameter.username ? e.parameter.username.trim() : '';
      var password = e.parameter.password ? e.parameter.password.trim() : '';
      
      var userSheet = ss.getSheetByName("Users");
      if (!userSheet) {
        return ContentService.createTextOutput(JSON.stringify({
          success: false,
          message: 'Sheet Users not found'
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var userData = userSheet.getDataRange().getValues();
      var authenticated = false;
      var userRole = 'Member';
      
      for (var i = 1; i < userData.length; i++) {
        var sheetUser = userData[i][0] ? userData[i][0].toString().trim() : '';
        var sheetPass = userData[i][1] ? userData[i][1].toString().trim() : '';
        
        if (sheetUser === username && sheetPass === password) {
          authenticated = true;
          userRole = userData[i][2] ? userData[i][2].toString().trim() : 'Member';
          break;
        }
      }
      
      var response = {
        success: authenticated,
        username: username,
        role: userRole,
        message: authenticated ? 'Login successful' : 'Invalid username or password'
      };
      
      return ContentService.createTextOutput(JSON.stringify(response))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var lessonId = e ? e.parameter.lessonId : null;

    // --- ACTION: SAVE VIDEO URL (บันทึกลิงก์วิดีโอ) ---
    if (action === 'saveVideoUrl') {
      var videoUrl = e ? e.parameter.videoUrl : null;
      var sheet = ss.getSheetByName("VideoLinks");
      
      if (!sheet) {
        sheet = ss.insertSheet("VideoLinks");
        sheet.appendRow(["Lesson ID", "Video URL", "Last Updated"]);
      }
      
      var timestamp = new Date();
      var data = sheet.getDataRange().getValues();
      var found = false;
      
      for (var i = 1; i < data.length; i++) {
        if (data[i][0] == lessonId) {
          sheet.getRange(i + 1, 2).setValue(videoUrl);
          sheet.getRange(i + 1, 3).setValue(timestamp);
          found = true;
          break;
        }
      }
      
      if (!found) {
        sheet.appendRow([lessonId, videoUrl, timestamp]);
      }
      
      return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
    }

    // --- ACTION: SAVE COVER URL (บันทึกลิงก์รูปภาพหน้าปก) ---
    if (action === 'saveCoverUrl') {
      var coverUrl = e ? e.parameter.coverUrl : null;
      var sheet = ss.getSheetByName("CoverLinks");
      
      if (!sheet) {
        sheet = ss.insertSheet("CoverLinks");
        sheet.appendRow(["Lesson ID", "Cover URL", "Last Updated"]);
      }
      
      var timestamp = new Date();
      var data = sheet.getDataRange().getValues();
      var found = false;
      
      for (var i = 1; i < data.length; i++) {
        if (data[i][0] == lessonId) {
          sheet.getRange(i + 1, 2).setValue(coverUrl);
          sheet.getRange(i + 1, 3).setValue(timestamp);
          found = true;
          break;
        }
      }
      
      if (!found) {
        sheet.appendRow([lessonId, coverUrl, timestamp]);
      }
      
      return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
    }

    // --- ACTION: SAVE FEED POST (บันทึกบทวิเคราะห์แผนการเทรดรายวัน) ---
    if (action === 'saveFeedPost') {
      var feedSheet = ss.getSheetByName("Feed");
      
      if (!feedSheet) {
        feedSheet = ss.insertSheet("Feed");
        feedSheet.appendRow(["ID", "DateStr", "Author", "Role", "Tag", "Content", "ImageUrl", "Timestamp"]);
      }
      
      var id = (e && e.parameter.id) ? e.parameter.id : ('post_' + new Date().getTime());
      var dateStr = (e && e.parameter.dateStr) ? e.parameter.dateStr : '';
      var author = (e && e.parameter.author) ? e.parameter.author : 'Sira';
      var role = (e && e.parameter.role) ? e.parameter.role : 'Editor';
      var tag = (e && e.parameter.tag) ? e.parameter.tag : '';
      var content = (e && e.parameter.content) ? e.parameter.content : '';
      var imageUrl = (e && e.parameter.imageUrl) ? e.parameter.imageUrl : '';
      var timestamp = (e && e.parameter.timestamp) ? e.parameter.timestamp : new Date().getTime();

      // จัดการรูปภาพ: หากแนบ Base64 มา ให้อัปโหลดลงโฟลเดอร์ "Feed" ใน Google Drive โดยตรง
      if (imageUrl && imageUrl.indexOf('data:image') === 0) {
        try {
          var parts = imageUrl.split(',');
          var mimeMatch = parts[0].match(/:(.*?);/);
          var mimeType = mimeMatch ? mimeMatch[1] : 'image/jpeg';
          var base64Data = parts[1].replace(/ /g, '+');
          var decoded = Utilities.base64Decode(base64Data);
          var blob = Utilities.newBlob(decoded, mimeType, 'chart_' + id + '.jpg');
          
          // บันทึกลงโฟลเดอร์ "Feed" ใน Google Drive โดยอัตโนมัติ (ไม่ใช้ ID ป้องกันพิมพ์ผิด)
          var folders = DriveApp.getFoldersByName("Feed");
          var targetFolder = folders.hasNext() ? folders.next() : DriveApp.getRootFolder();

          var file = targetFolder.createFile(blob);
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          imageUrl = 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w2000';
        } catch (driveErr) {
          imageUrl = 'DRIVE_ERR: ' + driveErr.toString();
        }
      }

      // แทรกแถวบนสุด (แถวที่ 2 ใต้หัวตาราง) เพื่อให้โพสต์ล่าสุดอยู่บนเสมอ
      if (feedSheet.getLastRow() <= 1) {
        feedSheet.appendRow([id, dateStr, author, role, tag, content, imageUrl, timestamp]);
      } else {
        feedSheet.insertRowBefore(2);
        feedSheet.getRange(2, 1, 1, 8).setValues([[id, dateStr, author, role, tag, content, imageUrl, timestamp]]);
      }
      
      return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
    }

    // --- ACTION: DELETE FEED POST (ลบบทวิเคราะห์ พร้อมลบรูปใน Google Drive และความคิดเห็น) ---
    if (action === 'deleteFeedPost') {
      var feedSheet = ss.getSheetByName("Feed");
      var deleteId = e ? e.parameter.id : null;
      if (feedSheet && deleteId) {
        var data = feedSheet.getDataRange().getValues();
        var targetDel = String(deleteId).trim();
        for (var i = 1; i < data.length; i++) {
          if (String(data[i][0]).trim() === targetDel) {
            // 1. ตรวจสอบและลบรูปภาพออกจาก Google Drive ทันที
            var imgUrl = String(data[i][6] || '');
            if (imgUrl) {
              var fileId = null;
              var idMatch = imgUrl.match(/id=([^&]+)/);
              if (idMatch) {
                fileId = idMatch[1];
              } else {
                var dMatch = imgUrl.match(/\/d\/([^/?]+)/);
                if (dMatch) fileId = dMatch[1];
              }
              if (fileId) {
                try {
                  var file = DriveApp.getFileById(fileId);
                  file.setTrashed(true); // ย้ายรูปภาพลงถังขยะใน Google Drive ทันที
                } catch (fileErr) {}
              }
            }

            // 2. ลบแถวโพสต์ออกจากตาราง Feed
            feedSheet.deleteRow(i + 1);
            break;
          }
        }
      }

      // 3. ลบความคิดเห็นที่ผูกกับโพสต์นี้ออกจากตาราง FeedComments (ถ้ามี)
      var commentSheet = ss.getSheetByName("FeedComments");
      if (commentSheet && deleteId) {
        var cData = commentSheet.getDataRange().getValues();
        var targetPostId = String(deleteId).trim();
        for (var c = cData.length - 1; c >= 1; c--) {
          if (String(cData[c][0]).trim() === targetPostId) {
            commentSheet.deleteRow(c + 1);
          }
        }
      }

      return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
    }

    // --- ACTION: SAVE FEED COMMENT (บันทึกความคิดเห็นใต้โพสต์) [เพิ่มใหม่] ---
    if (action === 'saveFeedComment') {
      var commentSheet = ss.getSheetByName("FeedComments");
      if (!commentSheet) {
        commentSheet = ss.insertSheet("FeedComments");
        commentSheet.appendRow(["PostID", "CommentID", "Author", "Role", "Content", "DateStr", "Timestamp"]);
      }
      var postId = (e && e.parameter.postId) ? e.parameter.postId : '';
      var commentId = (e && e.parameter.commentId) ? e.parameter.commentId : ('c_' + new Date().getTime());
      var author = (e && e.parameter.author) ? e.parameter.author : 'Member';
      var role = (e && e.parameter.role) ? e.parameter.role : 'Member';
      var content = (e && e.parameter.content) ? e.parameter.content : '';
      var dateStr = (e && e.parameter.dateStr) ? e.parameter.dateStr : '';
      var timestamp = (e && e.parameter.timestamp) ? e.parameter.timestamp : new Date().getTime();

      commentSheet.appendRow([postId, commentId, author, role, content, dateStr, timestamp]);
      return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
    }

    // --- ACTION: DELETE FEED COMMENT (ลบความคิดเห็น) [เพิ่มใหม่] ---
    if (action === 'deleteFeedComment') {
      var commentSheet = ss.getSheetByName("FeedComments");
      var delId = e ? e.parameter.commentId : null;
      var delPostId = e ? e.parameter.postId : null;
      var delContent = e ? e.parameter.content : null;
      if (commentSheet) {
        var data = commentSheet.getDataRange().getValues();
        var targetDel = delId ? String(delId).trim() : '';
        var targetPostId = delPostId ? String(delPostId).trim() : '';
        var targetContent = delContent ? String(delContent).trim() : '';
        for (var i = 1; i < data.length; i++) {
          var matchId = targetDel && String(data[i][1]).trim() === targetDel;
          var matchFallback = targetPostId && targetContent && String(data[i][0]).trim() === targetPostId && String(data[i][4]).trim() === targetContent;
          if (matchId || matchFallback) {
            commentSheet.deleteRow(i + 1);
            break;
          }
        }
      }
      return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
    }
    
    return ContentService.createTextOutput("No Action Found").setMimeType(ContentService.MimeType.TEXT);
  } catch (error) {
    return ContentService.createTextOutput("Error: " + error.toString()).setMimeType(ContentService.MimeType.TEXT);
  }
}

// ⚠️ ฟังก์ชันสำหรับเปิดหน้าต่างขออนุญาตสิทธิ์ Google Drive (ค้นหาชื่อโฟลเดอร์ Feed อัตโนมัติ ไม่ใช้ ID)
function authorizeDrive() {
  var folders = DriveApp.getFoldersByName("Feed");
  var targetFolder = folders.hasNext() ? folders.next() : DriveApp.getRootFolder();
  var testFile = targetFolder.createFile("auth_test.txt", "Permission Granted");
  testFile.setTrashed(true);
}
📍 ขั้นตอนที่ 3: เปิดสิทธิ์ Drive และ Deploy เวอร์ชันใหม่ (สำคัญมาก)
  1. กด บันทึก (Save / ไอคอนแผ่นดิสก์)
  2. (สำคัญ) ด้านบนเลือกฟังก์ชัน authorizeDrive แล้วกดปุ่ม เรียกใช้ (Run) 1 ครั้ง > ป๊อปอัปจะเด้งขึ้นมาทันที ให้กด ตรวจสอบสิทธิ์ (Review Permissions) > เลือกอีเมลของคุณ > ขั้นสูง (Advanced) > ไปที่... (ไม่ปลอดภัย) > กด อนุญาต (Allow) เพื่อเปิดสิทธิ์บันทึกภาพลง Google Drive
  3. กดปุ่มสีน้ำเงิน ทำให้ใช้งานได้ (Deploy) > จัดการการทำให้ใช้งานได้ (Manage Deployments)
  4. กดไอคอน ดินสอ (Edit) > เปลี่ยน เวอร์ชัน (Version) เป็น "เวอร์ชันใหม่" (New Version)
  5. กด ทำให้ใช้งานได้ (Deploy) เป็นอันเสร็จสิ้น!
